1	/** Resolves the host origin shared by iframe responses and App event subscriptions. */
2	export function resolveTargetOrigin(targetWindow, explicit) {
3	    if (explicit)
4	        return explicit;
5	    const hostOrigin = new URLSearchParams(targetWindow.location.search).get('hana-host-origin');
6	    if (hostOrigin)
7	        return hostOrigin;
8	    try {
9	        return new URL(targetWindow.document.referrer).origin;
10	    }
11	    catch {
12	        return '*';
13	    }
14	}
